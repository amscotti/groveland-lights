```go
// MaxPhotoBytes is the upload size limit (A-206/A-207).
const MaxPhotoBytes = 15 << 20 // 15 MB

var (
    ErrPhotoTooLarge    = errors.New("photo: above the size limit")
    ErrPhotoUnsupported = errors.New("photo: unsupported image format")
    ErrPhotoTooManyPx   = errors.New("photo: above the pixel limit")
)

// LimitBody wraps the request body with http.MaxBytesReader(MaxPhotoBytes).
// Call it BEFORE any multipart parsing; the overflow surfaces as
// ErrPhotoTooLarge from Photo.
func LimitBody(r *http.Request)

// UploadedPhoto is one persisted, processed photo.
type UploadedPhoto struct {
    ID          store.PhotoID
    Path        string // relative to PHOTO_DIR; URL path under /photos/
    Width, Height int
}

// Photo parses the optional "photo" file part through the img pipeline and
// persists it: PHOTO_DIR/YYYY/MM/<16-hex-rand>.jpg plus a photos row (relative
// path, dims, bytes, sha256 of the RE-ENCODED bytes). Returns (nil, nil) when
// the field is absent or empty (photo is optional). On ANY error nothing is
// left on disk and no row is written. Map: img.ErrTooLarge →
// ErrPhotoTooLarge; img.ErrUnsupported → ErrPhotoUnsupported (HEIC included);
// img.ErrPixelBomb → ErrPhotoTooManyPx.
func (s *Server) Photo(r *http.Request) (*UploadedPhoto, error)

// PhotoProblem maps an upload error to (httpStatus, userMessage):
//   ErrPhotoTooLarge    → 413, "Photo must be at most 15 MB"
//   ErrPhotoUnsupported → 415, "Photo must be JPEG, PNG or WebP — please choose JPEG instead of HEIC"
//                          when the bytes sniffed as HEIC (ftyp/heic), else
//                          415, "Photo must be JPEG, PNG or WebP"
//   ErrPhotoTooManyPx   → 415, "Photo has too many pixels"
func PhotoProblem(err error) (int, string)
```

Plus the serving route: `GET /photos/{path...}` — serves files from
`s.Cfg.PhotoDir` ONLY (resolve and verify the cleaned path stays inside
PhotoDir — traversal like `/photos/../../go.mod` must 404), `Content-Type:
image/jpeg`, `Cache-Control: public, max-age=31536000, immutable`. Missing → 404.

Rules frozen: options `img.Options{MaxBytes: MaxPhotoBytes, MaxPixels:
40_000_000, MaxEdge: 1600}`; directories created 0755, files 0644, both under
PHOTO_DIR only; filename server-generated (never client-controlled beyond
extension sniffing you ignore); sha256 over the RE-ENCODED bytes.
