/** Fresh per Playwright invocation; combined with the project code it makes
 * asset numbers unique across projects and across repeated runs against the
 * same persistent test-run.db. */
const RUN = Date.now().toString(36);

function projectCode(testInfo: TestInfo): string {
  switch (testInfo.project.name) {
    case 'desktop-chromium':
      return 'dc';
    case 'pixel-9':
      return 'p9';
    case 'iphone-16':
      return 'i16';
    default:
      return testInfo.project.name.replace(/[^a-z0-9]/g, '').slice(0, 4).toLowerCase() || 'x';
  }
}

/** Dense per-project index (0..3): separates the pagination seed lanes. */
function projectIndex(testInfo: TestInfo): number {
  switch (testInfo.project.name) {
    case 'desktop-chromium':
      return 0;
    case 'pixel-9':
      return 1;
    case 'iphone-16':
      return 2;
    default:
      return 3;
  }
}

/** `GP-017-<run>-<proj>-<tag>`: packet prefix preserved, value globally
 * unique PER TEST (the tag distinguishes @A-202/@A-205/@A-206/@A-208 even
 * when they share the GP-017 prefix), within the frozen 32-char
 * [A-Za-z0-9-] asset_no budget. */
function uniqAsset(prefix: string, testInfo: TestInfo, tag: string): string {
  return `${prefix}-${RUN}-${projectCode(testInfo)}-${tag}`;
}

/** Deterministic per-(run, key) coordinates inside TOWN_BBOX with a wide
 * margin: lat 42.7200..42.7500, lon -71.0800..-71.0500. Distinct keys land
 * (for practical purposes) never within the 8 m unnumbered-duplicate radius
 * of each other, so unnumbered creates cannot collide across tests, projects
 * or runs. */
function jittered(key: string): { lat: number; lon: number } {
  let h = 2166136261;
  for (const ch of `${RUN}|${key}`) {
    h = Math.imul(h ^ ch.charCodeAt(0), 16777619) >>> 0;
  }
  return {
    lat: 42.72 + (h % 300) * 0.0001,
    lon: -71.08 + ((h >>> 9) % 300) * 0.0001,
  };
}

/** The add-light form's CSRF token (double-submit: the matching cookie lands
 * in the shared context jar with this request). */
async function csrfToken(page: Page): Promise<string> {
  const res = await page.request.get('/admin/walk');
  expect(res.status(), 'GET /admin/walk while fetching csrf_token').toBe(200);
  const html = await res.text();
  const input = html.match(/<input[^>]*name="csrf_token"[^>]*>/)?.[0];
  const value = input?.match(/value="([^"]*)"/)?.[1];
  expect(value, 'csrf_token parsed from the walk page form').toBeTruthy();
  return value!;
}

/** Detail hrefs of every listed light whose row link mentions `label`
 * (frozen: the row link's accessible name contains the label), deduped —
 * a compliant row may carry the same href twice (thumbnail link + label
 * link), and cleanup must POST each delete exactly once. */
async function lightHrefsByLabel(page: Page, label: string): Promise<string[]> {
  const hrefs = new Set<string>();
  for (const target of ['/admin', '/admin?page=2']) {
    const res = await page.request.get(target);
    expect(res.status(), `GET ${target}`).toBe(200);
    for (const m of (await res.text()).matchAll(
      /<a\b[^>]*href="(\/admin\/lights\/\d+)"[^>]*>([\s\S]*?)<\/a>/g,
    )) {
      if (m[2].includes(label)) {
        hrefs.add(m[1]);
      }
    }
  }
  return [...hrefs];
}

/** Soft-delete one light by detail href (frozen: 303). */
async function deleteLight(page: Page, href: string, csrf: string): Promise<void> {
  const res = await page.request.post(`${href}/delete`, {
    multipart: { csrf_token: csrf },
    maxRedirects: 0,
  });
  expect(res.status(), `POST ${href}/delete`).toBe(303);
}

/** Delete every light whose row mentions `label`, sweeping until none
 * remain (deleting shifts later pages forward, so page-1 sweeps converge).
 * Used as pre-clean (stale rows from failed earlier runs / retries) and as
 * post-clean, keeping the shared DB tidy for the next test and the next
 * run. */
async function deleteAllByLabel(page: Page, label: string): Promise<void> {
  for (let sweep = 0; sweep < 6; sweep++) {
    const hrefs = await lightHrefsByLabel(page, label);
    if (hrefs.length === 0) {
      return;
    }
    const csrf = await csrfToken(page);
    for (const href of hrefs) {
      await deleteLight(page, href, csrf);
    }
  }
  throw new Error(`cleanup: lights labelled "${label}" survived 6 delete sweeps`);
}

/** Open the add-light form deterministically: wait for the FIRST geolocation
 * fix to populate the position readout (and the hidden lat/lon inputs)
 * before the form opens — a submit racing an empty position 422s. */
async function openWalkForm(page: Page): Promise<void> {
  await page.goto('/admin/walk');
  // #walk-pos is frozen at 6 decimal places (T-205) — assert that exact
  // form, not just "some decimals".
  await expect(page.locator('#walk-pos')).toHaveText(/\d+\.\d{6}, -?\d+\.\d{6}/);
  await expect(page.locator('#walk-accuracy')).toHaveText(/accuracy \d+ m/);
  await page.getByRole('button', { name: 'Add light here', exact: true }).click();
}

/** Walk the plan's §7 walk flow: open the add-light form and fill it (the
 * caller clicks `Confirm pin`, optionally after inspecting the panel). */
async function addLightFromForm(
  page: Page,
  opts: { assetNo?: string; label: string; photo?: string },
): Promise<void> {
  await openWalkForm(page);
  if (opts.assetNo) {
    await page.getByLabel('Asset number').fill(opts.assetNo);
  }
  await page.getByLabel('Label').fill(opts.label);
  await page.getByLabel('Kind').selectOption({ label: 'Park' });
  if (opts.photo) {
    await page.getByLabel('Photo').setInputFiles(opts.photo);
  }
}

test('@A-202 low accuracy gates saving behind an explicit save-anyway choice', async (
  { page, context },
  testInfo,
) => {
  const label = 'Low accuracy lamp';
  // This test SAVES an unnumbered light at 80 m accuracy, so its base
  // coordinates must be clear of every other test's/run's unnumbered
  // lights (jitter, not the shared default position).
  const base = jittered(`lowacc-${projectCode(testInfo)}`);
  await context.setGeolocation({ latitude: base.lat, longitude: base.lon, accuracy: 80 });
  await loginAsAdmin(page);
  await deleteAllByLabel(page, label); // rerun safety

  await page.goto('/admin/walk');

  await expect(page.locator('#walk-accuracy')).toHaveText(/accuracy 80 m/);
  const add = page.getByRole('button', { name: 'Add light here', exact: true });
  await expect(add).toBeVisible();
  // Over the 30 m threshold the primary action is disabled until the admin
  // explicitly accepts the low-accuracy risk. `Add light here` is the only
  // in-UI submit path, so while it is disabled NOTHING can be saved — and
  // the gate must be the platform's disabled ATTRIBUTE (asserted below),
  // which suppresses activation entirely; an aria-disabled look or a CSS
  // class alone would leave the click path live.
  await expect(add).toBeDisabled();
  await expect(add).toHaveAttribute('disabled');

  // Choosing save-anyway re-enables the button AND unlocks a REAL save at
  // 80 m — proving the gate is wired to the form, not a painted attribute.
  await page.getByRole('button', { name: 'Save anyway (low accuracy)' }).click();
  await expect(add).toBeEnabled();
  await add.click();
  await page.getByLabel('Label').fill(label);
  await page.getByRole('button', { name: 'Confirm pin' }).click();
  await expect(page.getByRole('status')).toContainText('Saved');
  await page.goto('/admin');
  await expect(page.locator(`img[alt="${label}"]`)).toBeVisible();

  await deleteAllByLabel(page, label);
});

test('@A-203 dragged pin position is what gets saved', async ({ page, context }, testInfo) => {
  const label = 'Dragged pin lamp';
  await loginAsAdmin(page);
  await deleteAllByLabel(page, label); // rerun safety (unnumbered light)

  // Jittered base coordinates: an unnumbered light 8 m from a previous
  // run's leftover would raise the duplicate panel instead of saving.
  const base = jittered(`drag-${projectCode(testInfo)}`);
  await context.setGeolocation({ latitude: base.lat, longitude: base.lon, accuracy: 10 });
  await page.goto('/admin/walk');

  const pin = page.locator('#pin');
  const pinPos = page.locator('#pin-pos');
  await expect(pin).toBeVisible();
  await expect(pin).toHaveAttribute('aria-label', 'position pin');
  await expect(pinPos).toHaveText(/^\d+\.\d{6}, -?\d+\.\d{6}$/);
  const readPos = async () => (await pinPos.innerText()).trim();
  const before = await readPos();

  // Drag the pin ~80 px; #pin-pos follows (possibly on the next frame, so
  // the read waits for the change instead of racing it).
  const box = (await pin.boundingBox())!;
  await page.mouse.move(box.x + box.width / 2, box.y + box.height / 2);
  await page.mouse.down();
  await page.mouse.move(box.x + box.width / 2 + 80, box.y + box.height / 2 + 80, {
    steps: 8,
  });
  await page.mouse.up();
  let afterDrag = '';
  await expect
    .poll(async () => {
      afterDrag = await readPos();
      return afterDrag;
    }, { message: '#pin-pos updates after the drag' })
    .not.toBe(before);

  // The frozen contract also makes #pin KEYBOARD-movable (arrow keys nudge
  // the effective position, the accessible alternative to dragging): nudge
  // east a few times and save THAT position, so the detail check below
  // covers the final keyboard-moved coordinates.
  await pin.focus();
  for (let i = 0; i < 3; i++) {
    await page.keyboard.press('ArrowRight');
  }
  let final = '';
  await expect
    .poll(async () => {
      final = await readPos();
      return final;
    }, { message: '#pin-pos updates after the arrow-key nudges' })
    .not.toBe(afterDrag);
  await expect(pinPos).toHaveText(/^\d+\.\d{6}, -?\d+\.\d{6}$/);

  await page.getByRole('button', { name: 'Add light here', exact: true }).click();
  await page.getByLabel('Label').fill(label);
  await page.getByRole('button', { name: 'Confirm pin' }).click();
  await expect(page.getByRole('status')).toContainText('Saved');

  // The admin detail shows exactly the confirmed coordinates — the same
  // 6 dp string #pin-pos rendered (both are frozen at 6 decimal places).
  await page.goto('/admin');
  await page.getByRole('link', { name: new RegExp(label) }).first().click();
  await expect(page).toHaveURL(/\/admin\/lights\/\d+$/);
  await expect(page.locator('#light-pos')).toHaveText(final);

  await deleteAllByLabel(page, label);
});

  { page, context },
  testInfo,
) => {
  const asset = uniqAsset('GP-017', testInfo, 'dup');
  const labels = ['Dup panel lamp', 'Dup panel lamp twin', 'Path light one', 'Path light two'];
  await loginAsAdmin(page);
  for (const label of labels) {
    await deleteAllByLabel(page, label);
  }

  // Numbered duplicate: the same asset_no posted again raises the panel.
  // Frozen (T-205): the numbered panel offers `Merge` + a cancel link

  await page.getByLabel('Asset number').fill(asset);

  await expect(panel.getByRole('heading', { name: 'Possible duplicate' })).toBeVisible();
  await expect(panel.getByRole('button', { name: 'Merge' })).toBeVisible();

  await page.getByRole('button', { name: 'Confirm pin' }).click();

  await expect(panel.getByRole('button', { name: 'Merge' })).toBeVisible();

test('@A-206 served photo is re-encoded, oriented and metadata-free', async (
  { page },
  testInfo,
) => {
  const asset = uniqAsset('GP-017', testInfo, 'photo'); // `img[alt^="GP-017"]` still matches
  const label = 'Photo purity lamp';
  await loginAsAdmin(page);
  await deleteAllByLabel(page, label);

  // Self-contained: this test creates its own photo-bearing light through
  // the same UI flow (make e2e-task must be able to run @A-206 alone).
  await addLightFromForm(page, {
    assetNo: asset,
    label,
    photo: 'testdata/photos/portrait-orient6-gps.jpg',
  });
  await page.getByRole('button', { name: 'Confirm pin' }).click();
  await expect(page.getByRole('status')).toContainText('Saved');

  await page.goto('/admin');
  const img = page.locator(`img[alt^="${asset}"]`);
  await expect(img).toBeVisible();
  const src = await img.getAttribute('src');
  expect(src).toBeTruthy();
  const res = await page.request.get(src!);
  expect(res.status()).toBe(200);
  const bytes = new Uint8Array(await res.body());
  // Purity is stronger than "no GPS": the served JPEG must carry NO APP1
  // segment at all (the Go draft asserts the same byte-level contract).
  expect(hasAPP1(bytes)).toBe(false);
  expect(hasExifGps(bytes)).toBe(false);
  const { w, h } = jpegSize(bytes);
  expect(Math.max(w, h)).toBeLessThanOrEqual(1600);
  expect(w).toBeGreaterThan(h); // 600x800 + Orientation=6 => wider than tall

  // The in-form preview renders the orientation already applied — assert
  // the INTRINSIC dimensions (naturalWidth/naturalHeight honour EXIF
  // orientation), not the CSS box, which a plain portrait stretched by a
  // wide container could game.
  await openWalkForm(page);
  await page.getByLabel('Label').fill(label);
  const preview = page.locator('#photo-preview');
  // A STATIC landscape placeholder must not pass: capture what #photo-preview
  // shows before the pick; whatever mechanism the page uses (object URL,
  // data URL, processed thumbnail), picking a file must change (or first
  // produce) the rendered source.
  const hadPreview = (await preview.count()) === 1;
  const srcBefore = hadPreview ? await preview.getAttribute('src') : null;
  await page.getByLabel('Photo').setInputFiles('testdata/photos/portrait-orient6-gps.jpg');
  await expect(preview).toBeVisible();
  // naturalWidth/naturalHeight are 0 until the preview image has LOADED —
  // CSS visibility alone does not imply decode — so poll for a decoded frame
  // before comparing. Browsers honour EXIF orientation in the intrinsic
  // dimensions (a processed or browser-oriented preview both qualify).
  let naturalW = 0;
  let naturalH = 0;
  await expect
    .poll(
      async () => {
        [naturalW, naturalH] = await preview.evaluate(
          (img: HTMLImageElement) => [img.naturalWidth, img.naturalHeight] as [number, number],
        );
        return naturalW;
      },
      { message: '#photo-preview decodes (naturalWidth > 0)' },
    )
    .toBeGreaterThan(0);
  expect(naturalW, 'preview intrinsic dims (EXIF orientation applied)').toBeGreaterThan(naturalH);
  if (hadPreview) {
    const srcAfter = await preview.getAttribute('src');
    expect(srcAfter, 'the preview must track the picked file (a static placeholder cannot)').not.toBe(srcBefore);
  }

  await deleteAllByLabel(page, label);
});

  await page.getByRole('button', { name: 'Delete light' }).click();

test('@A-209 admin list paginates with HTMX', async ({ page }, testInfo) => {
  const code = projectCode(testInfo);
  const baseLabel = `Scroll lamp ${RUN}-${code}`;
  await loginAsAdmin(page);
  // Rerun safety ACROSS INVOCATIONS: sweep the stable `Scroll lamp` prefix,
  // not just this run's label — a previously failed run's seeds ride the same
  // fixed longitude lane and would collide with this run's seeds (409).
  await deleteAllByLabel(page, 'Scroll lamp');

  // Seed 55 lights through the same HTTP contract the form uses (the UI
  // flow is covered above; 55 UI dialogs would take minutes). The seeds are
  // the 55 newest lights in the DB when the assertions run, so page 1 holds
  // exactly seeds 06..55 and page 2 holds 01..05 — regardless of any
  // leftovers from other tests. Unnumbered lights need coordinates clear of
  // the 8 m duplicate radius: lats ride a per-run jittered column spaced
  // ~55 m apart, while the lon is a FIXED per-project lane at
  // -71.0480..-71.0495 — outside the jitter band [-71.08,-71.05) and ~41 m
  // between lanes — so grid rows can never collide with each other across
  // projects or with any other test's jittered point, deterministically.
  const grid = jittered(`page-${code}`);
  const gridLon = -71.048 - projectIndex(testInfo) * 0.0005;
  const csrf = await csrfToken(page);
  for (let i = 1; i <= 55; i++) {
    const label = `${baseLabel} ${String(i).padStart(2, '0')}`;
    const res = await page.request.post('/admin/lights', {
      multipart: {
        csrf_token: csrf,
        label,
        kind: 'park',
        lat: (grid.lat + i * 0.0005).toFixed(7),
        lon: gridLon.toFixed(7),
      },
      maxRedirects: 0, // observe the frozen 303 instead of following it
    });
    expect(res.status(), `create ${label}`).toBe(303);
  }

  await page.goto('/admin');
  await expect(page.getByRole('heading', { name: 'Lights', exact: true })).toBeVisible();
  await expect(page.locator(`img[alt="${baseLabel} 55"]`)).toBeVisible();
  await expect(page.locator(`img[alt="${baseLabel} 01"]`)).toHaveCount(0);

  // "Older" swaps in page 2 as an HTMX fragment — no full navigation.
  let navigated = false;
  page.on('framenavigated', (frame) => {
    if (frame === page.mainFrame()) {
      navigated = true;
    }
  });
  const older = page
    .getByRole('link', { name: 'Older' })
    .or(page.getByRole('button', { name: 'Older' }));
  await older.click();
  await expect(page.locator(`img[alt="${baseLabel} 01"]`)).toBeVisible();
  await expect(page.locator(`img[alt="${baseLabel} 55"]`)).toHaveCount(0);
  expect(navigated).toBe(false);

  // Sweep the stable prefix again: this run's seeds AND any interleaved
  // leftovers must all be gone before the next invocation.
  await deleteAllByLabel(page, 'Scroll lamp');
});
